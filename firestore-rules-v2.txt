rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    match /newsletter_subscribers/{docId} {
      allow read, delete: if false;
      allow create, update:
        if request.resource.data.keys().hasOnly(['email', 'subscribedAt', 'uid'])
        && request.resource.data.email is string
        && request.resource.data.email.matches('^[^@\\s]+@[^@\\s]+[.][^@\\s]+$')
        && (request.resource.data.uid == null || request.resource.data.uid == request.auth.uid)
        && request.resource.size() < 2048;
    }

    match /shared_trips/{shareId} {
      allow read: if true;
      allow create:
        if request.auth != null
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.keys().hasOnly(
             ['name','countryId','countryName','flag','startDate','endDate','stops','transits','dailyNotes','ownerUid','updatedAt'])
        && request.resource.size() < 900000;
      allow update:
        if request.auth != null
        && resource.data.ownerUid == request.auth.uid
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.keys().hasOnly(
             ['name','countryId','countryName','flag','startDate','endDate','stops','transits','dailyNotes','ownerUid','updatedAt'])
        && request.resource.size() < 900000;
      allow delete:
        if request.auth != null && resource.data.ownerUid == request.auth.uid;
    }

  }
}
